        public static string najlepszyUczen(Dictionary<string,int> programowanie)
        {
            string nazwisko = "";
            int no = 0;

            foreach(var ocena in programowanie)
            {

                if(ocena.Value > no)
                {
                    no = ocena.Value;
                    nazwisko = ocena.Key;
                }
            }
            
            return nazwisko;
        }

        public static double sredniaOcen(Dictionary<string,int> programowanie)
        {
            double srednia = 0;
            double sumaOcen = 0;

            foreach (var ocena in programowanie)
            {
                sumaOcen += ocena.Value;

            }
                srednia = sumaOcen / programowanie.Count;

            return srednia;
        }

        public static string drugiNajlepszy(Dictionary<string, int> programowanie)
        {
            int najwyzsza = -1;
            int drugaNajwyzsza = -1;

            string najlepszyNazwisko = "";
            string drugiNajlepszyNazwisko = "";

            foreach (var ocena in programowanie)
            {
                if (ocena.Value > najwyzsza)
                {
                    drugaNajwyzsza = najwyzsza;
                    drugiNajlepszyNazwisko = najlepszyNazwisko;

                    najwyzsza = ocena.Value;
                    najlepszyNazwisko = ocena.Key;
                }
                else if (ocena.Value > drugaNajwyzsza)
                {
                    drugaNajwyzsza = ocena.Value;
                    drugiNajlepszyNazwisko = ocena.Key;
                }
            }

            return drugiNajlepszyNazwisko;
        }
