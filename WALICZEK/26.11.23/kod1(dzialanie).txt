        private bool PobierzLiczby(out double liczba1, out double liczba2, out double liczba3, out double liczba4)
        {
            bool poprawne = double.TryParse(txtLiczba1.Text, out liczba1) &
                double.TryParse(txtLiczba2.Text, out liczba2) &
                double.TryParse(txtLiczba3.Text, out liczba3) &
                double.TryParse(txtLiczba4.Text, out liczba4);
            {
                MessageBox.Show("Podaj poprawne liczby", "Info", MessageBoxButton.OK);
            }

            return poprawne;
        }

        
        //dodawanie
        private void Button_Click_1(object sender, RoutedEventArgs e)
        {
            if (PobierzLiczby(out double a, out double b, out double c, out double d))
            {
                Wynik.Text = (a + b).ToString();
            }
