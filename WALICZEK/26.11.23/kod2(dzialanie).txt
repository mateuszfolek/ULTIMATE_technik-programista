using System.Text;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Data;
using System.Windows.Documents;
using System.Windows.Input;
using System.Windows.Media;
using System.Windows.Media.Imaging;
using System.Windows.Navigation;
using System.Windows.Shapes;

namespace AnalizaTekstu
{
    /// <summary>
    /// Interaction logic for MainWindow.xaml
    /// </summary>
    public partial class MainWindow : Window
    {
        public MainWindow()
        {
            InitializeComponent();
        }

        public static int LiczbaZnaki(string tekst) => tekst.Length;
        public static int LiczbaSlowa(string tekst) =>
            tekst.Split((char[]?)null,StringSplitOptions.RemoveEmptyEntries).Length;

        public static int LiczbaSamogloski(string tekst) =>
            tekst.Count(c=>"aęeęuioyó".Contains(char.ToLowerInvariant(c)));

        public static int LiczbaWiersze(string tekst) =>
            tekst.Length == 0 ? 0 : tekst.Split('\n').Length;

        private void Analizuj_tekst(object sender, RoutedEventArgs e)
        {
            string t = txtTekst.Text;
            lblZnaki.Text = $"znaki: {LiczbaZnaki(t)}";
            lblwiersze.Text = $"Wiersze: {LiczbaWiersze(t)}";
            lblSlowa.Text = $"slowa: {LiczbaSlowa(t)}";
            lblSamogloski.Text = $"samogloski: {LiczbaSamogloski}";

        }
    }
}
